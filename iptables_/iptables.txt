# Allow loopback (VERY important) 
> sudo iptables -A INPUT -i lo -j ACCEPT


# allow port 1234 to recive packets
sudo iptables -A INPUT -p tcp --dport 1234 -j ACCEPT   # -p  for protocol 


Final structure (clean + production-style)

# allow good traffic
sudo iptables -A INPUT -i lo -j ACCEPT  # this for local (lo)
sudo iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT  # this allow initiated connection 

# allow your services (like using netcat or specific service or connection )
sudo iptables -A INPUT -p tcp --dport 1234 -j ACCEPT

# log the rest (rate-limited) to log droped packets
	> sudo iptables -A INPUT -m limit --limit 5/min -j LOG --log-prefix "DROP: "
	> sudo dmesg -C  this to clear  the kernel ring buffer that contain logs
        > sudo dmesg | grep DROP  (display the logs)



# drop everything else
method 1 : 
	> sudo iptables -P INPUT DROP  # drop as ploicy

method 2:
	> sudo iptables -A INPUT -j DROP # drop as rule
